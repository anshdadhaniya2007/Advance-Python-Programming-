
import threading
import time

# First task
def task1():
    for i in range(1, 4):
        print("Task 1: Downloading file", i)
        time.sleep(1)

# Second task
def task2():
    for i in range(1, 4):
        print("Task 2: Playing music", i)
        time.sleep(1)

# Third task
def task3():
    for i in range(1, 4):
        print("Task 3: Sending messages", i)
        time.sleep(1)

# Creating threads
t1 = threading.Thread(target=task1)
t2 = threading.Thread(target=task2)
t3 = threading.Thread(target=task3)

# Starting threads
t1.start()
t2.start()
t3.start()

# Waiting for all threads
t1.join()
t2.join()
t3.join()

print("All tasks completed!")
